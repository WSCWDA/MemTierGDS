cmake_minimum_required(VERSION 3.18)
project(memtier_uvm_opt LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(CUDAToolkit REQUIRED)
find_library(CUFILE_LIBRARY cufile)
find_path(CUFILE_INCLUDE_DIR cufile.h PATHS ${CUDAToolkit_INCLUDE_DIRS})
if(CUFILE_LIBRARY AND CUFILE_INCLUDE_DIR)
  set(MEMTIER_ENABLE_GDS 1)
  message(STATUS "cuFile found: ${CUFILE_LIBRARY}")
else()
  set(MEMTIER_ENABLE_GDS 0)
  message(STATUS "cuFile not found; building POSIX-only fallback")
endif()
add_library(memtier STATIC src/memtier.cpp src/cuda_utils.cpp src/object_table.cpp src/residency_table.cpp src/path_selector.cpp src/posix_backend.cpp src/gds_backend.cpp src/dram_cache.cpp)
target_include_directories(memtier PUBLIC include PRIVATE ${CUFILE_INCLUDE_DIR})
target_compile_definitions(memtier PUBLIC MEMTIER_ENABLE_GDS=${MEMTIER_ENABLE_GDS})
target_link_libraries(memtier PUBLIC CUDA::cudart pthread dl)
if(MEMTIER_ENABLE_GDS)
  target_link_libraries(memtier PUBLIC ${CUFILE_LIBRARY})
endif()
add_executable(simple_map_file examples/simple_map_file.cu)
target_link_libraries(simple_map_file PRIVATE memtier)
add_executable(compare_paths examples/compare_paths.cu)
target_link_libraries(compare_paths PRIVATE memtier)
add_executable(test_object_table tests/test_object_table.cpp)
target_link_libraries(test_object_table PRIVATE memtier)
add_executable(test_path_selector tests/test_path_selector.cpp)
target_link_libraries(test_path_selector PRIVATE memtier)
enable_testing()
add_test(NAME test_object_table COMMAND test_object_table)
add_test(NAME test_path_selector COMMAND test_path_selector)
